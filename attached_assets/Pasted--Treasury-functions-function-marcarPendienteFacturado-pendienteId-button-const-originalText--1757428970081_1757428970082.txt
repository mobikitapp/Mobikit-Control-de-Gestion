// Treasury functions function marcarPendienteFacturado(pendienteId, button) { const originalText = button.innerHTML; button.disabled = true; button.innerHTML = 'Procesando...'; const numeroFactura = prompt('Ingrese el número de factura (opcional):'); fetch(`/proyectos/pendientes-facturar/${pendienteId}/marcar-facturado`, { method: 'POST', headers: { 'Content-Type': 'application/json', }, body: JSON.stringify({ numero_factura: numeroFactura }) }) .then(response => response.json()) .then(data => { if (data.success) { showAlert('success', data.message); location.reload(); } else { showAlert('danger', 'Error: ' + data.message); button.disabled = false; button.innerHTML = originalText; } }) .catch(error => { console.error('Error:', error); showAlert('danger', 'Error de conexión. Por favor, intenta nuevamente.'); button.disabled = false; button.innerHTML = originalText; }); } function editarFechaProgramada(pendienteId, fechaActual) { // Create a modal for date picker const modalHtml = `

Nueva Fecha Programada

Cancelar Guardar
`; // Remove existing modal if any const existingModal = document.getElementById('editarFechaModal'); if (existingModal) { existingModal.remove(); } // Add modal to page document.body.insertAdjacentHTML('beforeend', modalHtml); // Show modal const modal = new bootstrap.Modal(document.getElementById('editarFechaModal')); modal.show(); } function guardarFechaProgramada(pendienteId) { const nuevaFecha = document.getElementById('nuevaFechaProgramada').value; if (!nuevaFecha) { showAlert('warning', 'Por favor seleccione una fecha'); return; } fetch(`/proyectos/pendientes-facturar/${pendienteId}/editar-fecha-programada`, { method: 'POST', headers: { 'Content-Type': 'application/json', }, body: JSON.stringify({ fecha_programada: nuevaFecha }) }) .then(response => response.json()) .then(data => { if (data.success) { showAlert('success', data.message); bootstrap.Modal.getInstance(document.getElementById('editarFechaModal')).hide(); location.reload(); } else { showAlert('danger', 'Error: ' + data.message); } }) .catch(error => { console.error('Error:', error); showAlert('danger', 'Error de conexión. Por favor, intenta nuevamente.'); }); } function definirFechaProgramada(pendienteId, button) { const originalText = button.innerHTML; // Create a modal for date picker const modalHtml = `

Fecha Programada de Facturación 
dd-mm-aaaa
Cancelar Definir Fecha
`; // Remove existing modal if any const existingModal = document.getElementById('definirFechaModal'); if (existingModal) { existingModal.remove(); } // Add modal to page document.body.insertAdjacentHTML('beforeend', modalHtml); // Show modal const modal = new bootstrap.Modal(document.getElementById('definirFechaModal')); modal.show(); } function confirmarDefinirFecha(pendienteId, originalText) { const fecha = document.getElementById('fechaProgramadaDefinir').value; if (!fecha) { showAlert('warning', 'Por favor seleccione una fecha'); return; } fetch(`/proyectos/pendientes-facturar/${pendienteId}/definir-fecha-programada`, { method: 'POST', headers: { 'Content-Type': 'application/json', }, body: JSON.stringify({ fecha_programada: fecha }) }) .then(response => response.json()) .then(data => { if (data.success) { showAlert('success', data.message); bootstrap.Modal.getInstance(document.getElementById('definirFechaModal')).hide(); location.reload(); } else { showAlert('danger', 'Error: ' + data.message); } }) .catch(error => { console.error('Error:', error); showAlert('danger', 'Error de conexión.
